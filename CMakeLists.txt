# Workspace dispatcher. Build exactly one engine per CMake tree.
cmake_minimum_required(VERSION 3.28)
project(EngineWorkspace LANGUAGES C CXX)
include(CTest)
set(ENGINE "" CACHE STRING "Engine to configure: retroforge or vestigio")
set_property(CACHE ENGINE PROPERTY STRINGS retroforge vestigio)
if(ENGINE STREQUAL "retroforge")
    add_subdirectory(engines/retroforge)
elseif(ENGINE STREQUAL "vestigio")
    add_subdirectory(engines/vestigio)
else()
    message(FATAL_ERROR "Select one engine: -DENGINE=retroforge or -DENGINE=vestigio. See CMakePresets.json.")
endif()
